Cleaning 10,000 messy customer records

A before/after data-quality case study in R

The problem

A customer dataset of 10,000 rows and 12 columns (from Kaggle) with mixed formats, placeholder value, impossible numbers and lot of missing data. The goal: clean it without inventing any values, and measure the improvement column by column.

Problems found

  • CustomerID: mixed format (C2 vs 3)
  • Gender: F, Female, female, male, Male
  • Age: impossible values (-5, 200)
  • Dates: stored as text (dd/mm/yyyy) with junk like not_a_date
  • purchase_amount: text, with -999 and N/A used as placeholders
  • feedback_score: -1 used as a “no answer” code
  • email: some missing the @
  • Phone_number: values like abc123 and 0

How I cleaned it

Code 
clean_2 <- messy_data |> 
  mutate(
    # Keep the number and standardise the "C" prefix
    customer_id = if_else(is.na(CustomerID), NA_character_,paste0("C",parse_number(CustomerID))),
    
    customer_name = str_to_title(Name),
    
    # standardise gender to Male / Female (Case-insensitive)
    gender = case_when(
      str_to_lower(Gender) %in% c("f", "female") ~ "Female",
      str_to_lower(Gender) %in% c("m", "male") ~ "Male",
      TRUE ~ NA_character_
    ),
    
    # impossible ages become missing
    age = if_else(Age < 0 | Age > 100, NA_real_, Age),
    
    city = str_to_title(City),
    
    # text date (dd/mm/yyyy) become real dates; junk becomes NA
    signup_date = dmy(Signup_Date, quiet = TRUE),
    last_purchase_date = dmy(Last_purchase_date, quiet = TRUE),
    
    # placeholders become missing, then convert to numeric
    purchase_amount_clean = parse_number(
      if_else(purchase_amount %in% c("N/A", "-999"), NA_character_, purchase_amount)
    ),
    # -1 means "no answer"
    feedback_score = if_else(feedback_score < 0, NA_real_, feedback_score),
    
    # repair emails missing the "@" before "mail.com"
    email_cleaned = case_when(
      is.na(email) ~ NA_character_,
      str_detect(email, "@") ~ email,
      str_detect(email, "mail\\.com$") ~ str_replace(email, "mail\\.com$", "@mail.com"),
      TRUE ~ NA_character_
    ),
    
    # phone numbers must be exactly 10 digits
    phone_number = if_else(str_detect(Phone_number, "^\\d{10}$"),
                           Phone_number, NA_character_),
    country = str_squish(Country)
  ) |> 
  select(customer_id, customer_name, gender, age, city, signup_date, last_purchase_date, purchase_amount_clean, feedback_score, email_cleaned, phone_number, country) |> 
  rename(purchase_amount = purchase_amount_clean)

Assumptions

  • Invalid values become NA. Nothing is guessed or filled in, and no rows are dropped.
  • -1 in feedback_score is treated as “no answer”. Scores of 10 are kept as valid, although the rest of the scale looks like 1 to 5.
  • A broken email is assumed to be missing only the ‘@’ before mail.com
  • Dates are read as day/month/year.

Validation

Code 
email_pattern <- "^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$"

checks <- tribble(
  ~check,                                  ~passed,
  "Row count preserved",                   nrow(clean_2) == nrow(messy_data),
  "No double @ in emails",                 sum(str_count(clean_2$email_cleaned, "@") > 1, na.rm = TRUE) == 0,
  "All emails match a valid pattern",      all(str_detect(clean_2$email_cleaned, email_pattern), na.rm = TRUE),
  "Gender only Male / Female / NA",        all(clean_2$gender %in% c("Male", "Female", NA)),
  "Age within 0 to 100",                   all(clean_2$age >= 0 & clean_2$age <= 100, na.rm = TRUE),
  "purchase_amount numeric and positive",  is.numeric(clean_2$purchase_amount) &&
                                           sum(!is.na(clean_2$purchase_amount)) > 0 &&
                                           all(clean_2$purchase_amount > 0, na.rm = TRUE),
  "Phones are 10 digits",                  all(str_detect(clean_2$phone_number, "^\\d{10}$"), na.rm = TRUE)
) |>
  mutate(result = if_else(passed, "PASS", "FAIL")) |>
  select(check, result)

Before and After

Code 
count_status <- function(x, is_valid) {
  is_miss <- is.na(x)
  is_ok <- !is_miss & coalesce(is_valid, FALSE)
  is_bad <- !is_miss & !is_ok
  tibble(valid = sum(is_ok), missing = sum(is_miss), invalid = sum(is_bad))
}

before <- bind_rows(
  count_status(messy_data$CustomerID, str_detect(messy_data$CustomerID, "^C\\d+$")) |> mutate(column = "customer_id"),
  count_status(messy_data$Gender, messy_data$Gender %in% c("Male", "Female")) |> mutate(column = "gender"),
  count_status(messy_data$Age, messy_data$Age >= 0 & messy_data$Age <= 100) |> mutate(column = "age"),
  count_status(messy_data$Signup_Date, !is.na(dmy(messy_data$Signup_Date, quiet = TRUE))) |> mutate(column = "signup_date"),
  count_status(messy_data$Last_purchase_date, !is.na(dmy(messy_data$Last_purchase_date, quiet = TRUE))) |> mutate(column = "last_purchase_date"),
  count_status(messy_data$purchase_amount, !messy_data$purchase_amount %in% c("-999", "N/A")) |> mutate(column = "purchase_amount"),
  count_status(messy_data$feedback_score, messy_data$feedback_score >= 0) |> mutate(column = "feedback_score"),
  count_status(messy_data$email, str_detect(messy_data$email, email_pattern)) |> mutate(column = "email"),
  count_status(messy_data$Phone_number, str_detect(messy_data$Phone_number, "^\\d{10}$")) |> mutate(column = "phone")
) |> 
  mutate(stage = "Before")


after <- bind_rows(
  count_status(clean_2$customer_id, str_detect(clean_2$customer_id, "^C\\d+$")) |> mutate(column = "customer_id"),
  count_status(clean_2$gender, clean_2$gender %in% c("Male", "Female")) |> mutate(column = "gender"),
  count_status(clean_2$age, clean_2$age >= 0 & clean_2$age <= 100) |> mutate(column = "age"),
  count_status(clean_2$signup_date, TRUE) |> mutate(column = "signup_date"),
  count_status(clean_2$last_purchase_date, TRUE) |> mutate(column = "last_purchase_date"),
  count_status(clean_2$purchase_amount, clean_2$purchase_amount > 0) |> mutate(column = "purchase_amount"),
  count_status(clean_2$feedback_score, clean_2$feedback_score >= 0) |> mutate(column = "feedback_score"),
  count_status(clean_2$email_cleaned, str_detect(clean_2$email_cleaned, email_pattern)) |> mutate(column = "email"),
  count_status(clean_2$phone_number, str_detect(clean_2$phone_number, "^\\d{10}$")) |> mutate(column = "phone")
) |> mutate(stage = "After")


scorecard <- bind_rows(before, after) |> 
  pivot_longer(c(valid, missing, invalid), names_to = "status", values_to = "n") |> 
  mutate(status = factor(status, levels = c("invalid", "missing", "valid")),
         stage = factor(stage, levels = c("Before", "After")))

# every column and stage must add up to the full row count
stopifnot(
  scorecard |> group_by(stage, column) |> 
    summarise(total = sum(n), .groups = "drop") |> 
    pull(total) |> unique() == nrow(messy_data)
)

Cleaning turns invalid values into missing ones, so “missing” goes up while “invalid” drops to zero. That is deliberate: missing value is honest, a guess one is not.

Findings

Code 
amount_missing <- mean(is.na(clean_2$purchase_amount))

date_check <- clean_2 |> 
  filter(!is.na(signup_date), !is.na(last_purchase_date)) |> 
  summarise(testable = n(),
            before_signup = sum(last_purchase_date < signup_date),
            pct = round(100 * before_signup / testable, 1))

  • Purchase amounts are mostly likely missing: 75% of rows have no usable amount, so any average spend rests on a minority of customers.

  • Dates don’t add up: among the 1,684 rows with both dates, 829 (49.2%) show a last purchase before signup. I flagged this and kept the rows. The cause is unknown (entry errors, migrated accounts or swapped fields).

  • City and country don’t match: every city is in India, but countries includes USA, UK, Canada and Australia.

Limitations

  • Only about 17% of rows have both dates, so the date finding rests on a small subset.
  • The rules reflect my assumptions about placeholders and formats, listed above.
  • The dataset looks synthetic, so findings illustrate the method and not real customers.
