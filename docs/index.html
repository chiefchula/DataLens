Cleaning ————————————-

clean <- messy_data |> mutate(across(where(is.character), str_squish)) |> mutate( customer_id = parse_number(CustomerID), gender = case_when( str_to_lower(Gender) %in% c(“f”, “female”) ~ “Female”, str_to_lower(Gender) %in% c(“m”, “male”) ~ “Male”, TRUE ~ NA_character_ ), age = if_else(Age >= 0 & Age <= 100, Age, NA_real_), signup_date = suppressWarnings(dmy(Signup_Date)), last_purchase_date = suppressWarnings(dmy(Last_purchase_date)), purchase_amount_clean = if_else(purchase_amount > 0, as.numeric(purchase_amount), NA_real_), feedback_score = if_else(feedback_score >= 0, feedback_score, NA_real_), email_valid = str_detect(email, “1+@[^@\\s]+\.[^@\\s]+\("), email = if_else(email_valid, str_to_lower(email), NA_character_), phone_valid = str_detect(Phone_number, "^\\d{10}\)”), phone = if_else(phone_valid, Phone_number, NA_character_) ) |> select(customer_id, name = Name, gender, age, city = City, signup_date, last_purchase_date, purchase_amount_clean, feedback_score, email, phone, country = Country)

clean_2 <- messy_data |> mutate( # pull the number from customerID and append “C” customer_id = if_else(is.na(CustomerID), NA_character_, paste0(“C”,parse_number(CustomerID))),

# Title capitalisation of names and the title
customer_name = str_to_title(Name),

# standardize all gender into two Male and Female
gender = case_when(
  Gender %in% c("female", "f") ~ "Female",
  Gender %in% c("male", "m") ~ "Male",
  TRUE ~ NA_character_
),

# Age to be between 0 and 100
age = if_else(Age < 0 | Age > 100, NA_real_, Age),

# Title capitalization to city
city = str_to_title(City), 

# Convert signup date and last purchase data as proper dates format
signup_date = dmy(Signup_Date, quiet = T),
last_purchase_date = dmy(Last_purchase_date, quiet = T),

# purchase amount to numeric, change N/A and -999 as missing values
purchase_amount_clean = parse_number(if_else(purchase_amount %in% c("N/A", "-999"), NA_character_, purchase_amount)),

# change -1 to NA
feedback_score = if_else(feedback_score < 0, NA_real_, feedback_score),

# email standardization
email_cleaned = case_when(
  is.na(email) ~ NA_character_,
  str_detect(email, "@") ~ email,                              # already valid, leave alone
  str_detect(email, "mail\\.com$") ~ str_replace(email, "mail\\.com$", "@mail.com"),  # insert the missing @
  TRUE ~ NA_character_
),

# stanadardize length of phone numbers
phone_number = if_else(str_detect(Phone_number, "^\\d{10}$"), Phone_number, NA_character_),

# Standardize country
country = str_squish(Country)

) |>

# retain only clean columns select(customer_id, customer_name, gender, age, city, signup_date, last_purchase_date, purchase_amount_clean, feedback_score, email_cleaned, phone_number, country) |> rename(purchase_amount = purchase_amount_clean)

count_status <- function(x, is_valid) { is_miss <- is.na(x) is_ok <- !is_miss & coalesce(is_valid, FALSE) is_bad <- !is_miss & !is_ok tibble(valid = sum(is_ok), missing = sum(is_miss), invalid = sum(is_bad)) }

email_ok <- “2+@[^@\\s]+\.[^@\\s]+$”

before <- bind_rows( count_status(messy_data\(CustomerID, str_detect(messy_data\)CustomerID, “^C\d+\(")) |> mutate(column = "customer_id"), count_status(messy_data\)Gender, messy_data\(Gender %in% c("Male", "Female")) |> mutate(column = "gender"), count_status(messy_data\)Age, messy_data\(Age >= 0 & messy_data\)Age <= 100) |> mutate(column =”age”), count_status(messy_data\(Signup_Date, !is.na(dmy(messy_data\)Signup_Date, quiet = TRUE))) |> mutate(column = “signup_date”), count_status(messy_data\(Last_purchase_date, !is.na(dmy(messy_data\)Last_purchase_date, quiet = TRUE))) |> mutate(column = “last_purchase_date”), count_status(messy_data\(purchase_amount, !messy_data\)purchase_amount %in% c(“-999”, “N/A”)) |> mutate(column = “purchase_amount”), count_status(messy_data\(feedback_score, messy_data\)feedback_score >= 0) |> mutate(column = “feedback_score”), count_status(messy_data\(email, str_detect(messy_data\)email, email_ok)) |> mutate(column = “email”), count_status(messy_data\(Phone_number, str_detect(messy_data\)Phone_number, “^\d{10}$”)) |> mutate(column = “phone”) ) |> mutate(stage = “Before”)

after <- bind_rows( count_status(clean_2\(customer_id, str_detect(clean_2\)customer_id, “^C\d+\(")) |> mutate(column = "customer_id"), count_status(clean_2\)gender, clean_2\(gender %in% c("Male", "Female")) |> mutate(column = "gender"), count_status(clean_2\)age, clean_2\(age >= 0 & clean_2\)age <= 100) |> mutate(column =”age”), count_status(clean_2\(signup_date, TRUE) |> mutate(column = "signup_date"), count_status(clean_2\)last_purchase_date, TRUE) |> mutate(column = “last_purchase_date”), count_status(clean_2\(purchase_amount, clean_2\)purchase_amount > 0) |> mutate(column = “purchase_amount”), count_status(clean_2\(feedback_score, clean_2\)feedback_score >= 0) |> mutate(column = “feedback_score”), count_status(clean_2\(email_cleaned, str_detect(clean_2\)email_cleaned, email_ok)) |> mutate(column = “email”), count_status(clean_2\(phone_number, str_detect(clean_2\)phone_number, “^\d{10}$”)) |> mutate(column = “phone”) ) |> mutate(stage = “After”)

scorecard <- bind_rows(before, after) |> pivot_longer(c(valid, missing, invalid), names_to = “status”, values_to = “n”) |> mutate(status = factor(status, levels = c(“invalid”, “missing”, “valid”)), stage = factor(stage, levels = c(“Before”, “After”)))

p <- scorecard |> ggplot(aes(stage, n, fill = status)) + geom_col() + geom_text( aes(label = if_else(n > 0, scales::comma(n), ““)), position = position_stack(vjust = 0.5), size = 3, colour =”white” ) + facet_wrap(~column, ncol = 3) + scale_fill_manual(values = c(valid = “#2e7d32”, missing = “#757575”, invalid = “#c62828”)) + labs(title = “Data quality before vs after cleaning”, x = NULL, y = “Rows”, fill = NULL) + theme_minimal() + theme(legend.position = “bottom”)
